import pandas as pd

data = {
    'Student': ['Alice', 'Bob', 'Charlie', 'David', 'Eve'],
    'Course1': [78, 85, 90, 60, 70],
    'Course2': [80, 75, 88, 92, 65],
    'Course3': [70, 65, 75, 80, 85],
    'Course4': [85, 90, 92, 78, 80]
}
df = pd.DataFrame(data)
print(df)

course2_marks = df['Course2']
print("\nCourse 2 marks:\n", course2_marks)

xyz_marks = df[df['Student'] == 'xyz']
print("\nXYZ marks:\n", xyz_marks)

alice_course1_course3 = df[df['Student'] == 'Alice'][['Course1', 'Course3']]
print("\nAlice marks for Course1 & Course3:\n", alice_course1_course3)

df['Total'] = df[['Course1', 'Course2', 'Course3', 'Course4']].sum(axis=1)
df['Percentage'] = (df['Total'] / 4).round(2)
print("\n", df)

pass_course1 = df[df['Course1'] > 50]['Student']
print("\nStudents with >50 in Course1:\n", pass_course1)

def pass_fail(percentage):
    return 'Pass' if percentage >= 50 else 'Fail'

df['Result'] = df['Percentage'].apply(pass_fail)
coursewise_marks = df.set_index('Student').drop(
    ['Total', 'Percentage', 'Result'], axis=1
)
print("\nCoursewise Marks:\n", coursewise_marks)

df.drop('Percentage', axis=1, inplace=True)

project_marks = pd.DataFrame({
    'Student': ['Alice', 'Bob', 'Charlie', 'David', 'Eve'],
    'Project_Marks': [90, 85, 88, 75, 80]
})
df = pd.merge(df, project_marks, on='Student')

sorted_students = df.sort_values(by='Total', ascending=False)
print("\nSorted by Total:\n", sorted_students)

df['Rank'] = df['Total'].rank(method='dense', ascending=False).astype(int)
print("\nWith Rank:\n", df)

coursewise_total = df[['Course1', 'Course2', 'Course3', 'Course4']].sum()
coursewise_avg = df[['Course1', 'Course2', 'Course3', 'Course4']].mean()
print("\nCoursewise Total:\n", coursewise_total)
print("\nCoursewise Average:\n", coursewise_avg)

# Aim: Write a code to read data from different file formats like HTML, CSV, JSON, XML
#      and check for missing data and outliers and handling them.

import pandas as pd

def read_data(file_path, file_format):
    if file_format.lower() == 'csv':
        df = pd.read_csv(file_path)
    elif file_format.lower() == 'xml':
        df = pd.read_xml(file_path)
    elif file_format.lower() == 'html':
        df = pd.read_html(file_path)[0]
    elif file_format.lower() == 'json':
        df = pd.read_json(file_path)
    else:
        raise ValueError("Unsupported file format. Please choose from json, xml, html, csv.")
    return df

def handle_missing_data(df):
    missing_values = df.isnull().sum()
    if missing_values.sum() > 0:
        print("Missing values found.")
        print(missing_values)
        # fill the missing value with mean
        numeric_cols = df.select_dtypes(include="number").columns
        df[numeric_cols] = df[numeric_cols].fillna(df[numeric_cols].mean())
    else:
        print("No missing values found.")
    return df

def handle_outliers(df):
    numeric_cols = df.select_dtypes(include="number").columns
    for col in numeric_cols:
        Q1 = df[col].quantile(0.25)
        Q3 = df[col].quantile(0.75)
        IQR = Q3 - Q1
        lower = Q1 - 1.5 * IQR
        upper = Q3 + 1.5 * IQR
        df = df[(df[col] >= lower) & (df[col] <= upper)]
    if df.empty:
        print("Warning: All rows were removed as outliers.")
    return df

file_path = 'Iris_Data.csv'
file_format = 'csv'

data = read_data(file_path, file_format)
print("\nBefore performing operations:\n", data)

handle_missing_data(data)
print("\nAfter handling missing values:\n", data)

data = handle_outliers(data)
print("\nAfter handling outliers:\n", data)

# Aim: Perform Reshaping of the hierarchical data and pivoting data frame data.

import pandas as pd

data = {
    ('A', 'a'): [1, 2, 3],
    ('A', 'b'): [4, 5, 6],
    ('B', 'a'): [7, 8, 9],
    ('B', 'b'): [10, 11, 12]
}
df = pd.DataFrame(data, index=['x', 'y', 'z'])

pivot_data = {
    'Date': ['2022-01-01', '2022-01-02', '2022-01-03', '2022-01-01', '2022-01-02', '2022-01-03'],
    'Category': ['A', 'A', 'A', 'B', 'B', 'B'],
    'Value': [10, 20, 30, 40, 50, 60]
}
pivot_df = pd.DataFrame(pivot_data)

# Reshaping hierarchical data
stacked_df = df.stack(future_stack=True)   # Stack
unstacked_df = stacked_df.unstack()        # Unstack

# Pivoting a DataFrame
pivoted_df = pivot_df.pivot(index='Date', columns='Category', values='Value')

print("Original DataFrame:\n")
print(df)
print("\nStacked DataFrame:\n")
print(stacked_df)
print("\nUnstacked DataFrame:\n")
print(unstacked_df)
print("\nPivoted DataFrame:\n")
print(pivoted_df)

=== 3. Level of detail (LOD) calculations ===
Syntax pattern:
   {FIXED|INCLUDE|EXCLUDE [Dim 1],[Dim 2] : AGG([Row-Level])}
 - FIXED, INCLUDE or EXCLUDE is a keyword that indicates the type of LOD calculation.
 - Dim 1, Dim 2 are a comma-separated list of dimension fields that define the level of detail
   at which the calculation will be performed. You may use any number of dimensions.
 - AGG is the aggregate function you wish to perform (such as SUM, AVG, MIN, and MAX).
   Row-Level is the row-level field or row-level calculation that will be aggregated as
   specified by the aggregation you choose.
Steps:
1. Create a Calculated Field as:
   Average Rent(exclude Room) -> {EXCLUDE [Room] : AVG(Rent)}
2. Place Measure Names in Columns and Building and Room in Rows.
3. Put Measure Values on Marks (Text) and remove all values except AVG(Rent) and
   Average Rent(exclude Room).

 - Select the option required (Running Total, Difference, Percent Difference, Percent of Total,

   YTD Growth) and the selected option will be applied on the visual.

=== Parameters ===
Creating a Parameter - there are multiple ways in Tableau:
 - Use the drop-down menu next to Dimensions in the data pane and select Create Parameter.
 - Right-click an empty area in the data pane and select Create Parameter.
 - Use the drop-down menu on a field, set, or parameter in the data pane and select Create Parameter...

Create Parameter dialog fields: Name, Properties (Data type, Display format, Current value,
Value when workbook opens), Allowable values (All / List / Range).

 - When the parameter is created, it appears in the data pane under the Parameters section.
   The drop-down menu for a parameter reveals an option, Show Parameter Control, which adds
   the parameter control to the view.
 - The parameter control can be shown as a single value list (e.g. "Select an Option":
   First Option / Second Option / Third Option).
 - This control can be shown on any sheet or dashboard and allows the end user to select a
   single value. When the value is changed, any calculations, filters, sets, or bins that use
   the parameter will be re-evaluated, and any views that are affected will be redrawn.

Practical No. 9
Aim: Create Geospatial feature maps in Tableau using Geospatial Data.
(No code - Tableau steps)

 - Geographic visualization is incredibly valuable when you need to understand where things
   happen and whether there are any spatial relationships within the data.
 - Tableau offers three main types of geographic visualization: Filled maps, Symbol maps, Density maps.

=== Filled Maps ===
Filled maps fill areas such as countries, states, counties, or ZIP codes to show a location.
The color that fills the area can be used to encode values, most often of aggregated measures
but sometimes also dimensions.
Goal: understand sales for Superstore and see whether there are any patterns geographically.
 - Create a new sheet "Sales by State".
 - Double-click the State field in the data pane. Tableau automatically creates a geographic
   visualization using the Latitude (generated), Longitude (generated), and State fields.
 - Drag the Sales field from the data pane and drop it on the Color shelf on the Marks card.
 - If the colors are not shown, check for the error in the bottom-right side of the map.
 - If any error like "49 unknown" occurs, click on it > Edit > change the country to United States.
 - Based on the fields and shelves used, Tableau switches the automatic mark type to Map.

=== Symbol Maps ===
In symbol maps, marks on the map are not drawn as filled regions; rather, marks are shapes or
symbols placed at specific geographic locations. The size, color, and shape may also be used to
encode additional dimensions and measures.
Goal: create a symbol map to show sales by postal code for the Superstore dataset.
Steps:
 - Create a new sheet "Sales by postal code" (open Superstore sales dataset.csv).
 - Double-click Postal Code under Dimensions. Tableau automatically adds Postal Code to the
   Detail of the Marks card and Longitude (generated) and Latitude (generated) to Columns and Rows.
 - The mark type is set to a circle by default, and a single circle is drawn for each postal code
   at the correct latitude and longitude.
 - Drag Sales from Measures to the Size shelf on the Marks card. Each circle is sized according
   to the sum of sales for that postal code.
 - Drag Profit from Measures to the Color shelf on the Marks card. This encodes the mark color
   to correspond to the sum of profit (Edit Colors SUM(Profit): Orange-Blue Diverging palette,
   Stepped Color 6 steps, Center 0).
 - Do some fine-tuned adjustments to the size and color by clicking on respective marks.
 - If any error like "49 unknown" occurs, click on it > Edit > change the country to United States.

=== Density Maps ===
Density maps show the spread and concentration of values within a geographic area. Instead of
individual points or symbols, the marks blend together, showing intensity in areas with a high
concentration.
Goal: understand the geographic concentration of orders.
 - Navigate to the "Density of Orders" sheet.
 - Double-click the Postal Code field in the data pane. Just as before, Tableau automatically
   creates a symbol map geographic visualization using the Latitude (generated),
   Longitude (generated), and State fields.
 - Using the drop-down menu on the Marks card, change the mark type to Density. The individual
   circles now blend together showing concentrations.
 - This density map displays a high concentration of orders from the East Coast.

=== Creating the Dashboard framework ===
At this point, we have all of the necessary views to achieve the objectives for our dashboard.
Now, all that remains is to arrange them and enable the interactivity that's required to
effectively tell the story:

1. Create a new dashboard by clicking the New Dashboard tab to the right of all existing worksheet
   tabs or by selecting Dashboard | New Dashboard from the menu.
2. Rename the new dashboard as "Is Least Profitable Always Unprofitable?".
3. At the bottom of the left sidebar, check Show dashboard title.
4. Add the views to the dashboard by dragging them from the Dashboard pane of the left sidebar and
   dropping them into the dashboard canvas. Arrange them: Overall Profit By Category and
   Top 10 Least Profitable Items on top; Profit by State and Profit Trend below.
   After adding views, take some time to reposition and resize various elements and views.
5. Use the drop-down menu on the Department filter and change the control to a Single Value (dropdown).
6. You'll notice that changing the value of the filter only changes the Overall Profit by Category
   view. You can adjust which views the filter applies to by using the drop-down menu.
   Select Apply to Worksheets | All Using This Data Source.
7. From the left sidebar, drag and drop a Text object above Overall Profit by Category and enter
   the following instructions:
      1. Select a Department from the dropdown
      2. Select a category below
      3. Select an Item below
8. Using the grip, move the Department filter immediately above the Top 10 Least Profitable Items view.
9. Size the text object to align the Top 10 view with the Overall view.
10. Move the Profit color legend below the Profit by State view.
11. Use the drop-down menu of Overall Profit by Category to Fit | Entire View. This will ensure
    that all of the categories are visible without the need for a scrollbar.
12. Additionally, fit the Top 10 Least Profitable Items to Entire View.

=== Implementing actions to guide the story ===
To enable interactivity, we'll use actions.

1. Click the Use as Filter button on the Overall Profit by Category view. This will cause the view
   to be used as an interactive filter for the entire dashboard. That is, when the user selects a
   bar, all other views will be filtered based on the selection.
2. From the main menu, select Dashboard | Actions. You'll see a list containing one action named
   Filter 1 (generated). This is the action that was created when you selected Use as Filter previously.
3. Click the Add Action > button and select Filter. The resulting dialog gives you options for
   selecting the source and target, as well as additional options for the action.
4. Here, we want an action that filters everything except the Overall Profit by Category view when
   the user selects an item. In the Add Filter Action dialog, set Source Sheets to Top 10 Least
   Profitable Items, and Target Sheets to Profit by State and Profit Trend. Make sure that the
   action is set to run on Select. Name the filter "Filter by Item", and then click OK on this
   dialog. Do the same on the Actions dialog.

You now have three filters (two are actions) that drive the dashboard:
 - Selecting a Department from the drop-down will filter the entire dashboard (and actually all
   views in the workbook as you set it to filter every view using the data source).
 - Selecting a Category (clicking a bar or header) will filter the entire dashboard to that selection.
 - Selecting an Item (clicking a bar or header) will filter the Profit by State and Profit Trend views.

Experiment with the filters and actions to see how your dashboard functions.
