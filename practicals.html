import hashlib
import secrets

def hash_password(password):
    salt = secrets.token_bytes(16)  # Generate a random 16-byte salt

    # Hash the password using PBKDF2-HMAC-SHA256
    password_hash = hashlib.pbkdf2_hmac(
        'sha256',
        password.encode('utf-8'),
        salt,
        100_000  # Number of iterations
    )

    return salt, password_hash

def verify_password(password, stored_salt, stored_hash):
    new_hash = hashlib.pbkdf2_hmac(
        'sha256',
        password.encode('utf-8'),
        stored_salt,
        100_000
    )

    return secrets.compare_digest(new_hash, stored_hash)

import string
import random

def generate_key():
    alphabet = list(string.ascii_uppercase)
    shuffled = alphabet.copy()
    random.shuffle(shuffled)

    # Create encryption and decryption dictionaries
    encrypt_key = dict(zip(alphabet, shuffled))
    decrypt_key = {v: k for k, v in encrypt_key.items()}

    return encrypt_key, decrypt_key

def encrypt(message, key):
    result = ""
    for char in message:
        if char.upper() in key:
            encrypted = key[char.upper()]
            result += encrypted if char.isupper() else encrypted.lower()
        else:
            result += char
    return result

def decrypt(ciphertext, key):
    result = ""
    for char in ciphertext:
        if char.upper() in key:
            decrypted = key[char.upper()]
            result += decrypted if char.isupper() else decrypted.lower()
        else:
            result += char
    return result

def generate_key(message, keyword):
    key = []
    keyword = keyword.upper()
    j = 0
    for char in message:
        if char.isalpha():
            key.append(keyword[j % len(keyword)])
            j += 1
        else:
            key.append(char)
    return "".join(key)

def encrypt(message, keyword):
    key = generate_key(message, keyword)
    ciphertext = ""
    for m, k in zip(message, key):
        if m.isalpha():
            shift = ord(k) - ord('A')
            if m.isupper():
                encrypted = chr((ord(m) - ord('A') + shift) % 26 + ord('A'))
            else:
                encrypted = chr((ord(m) - ord('a') + shift) % 26 + ord('a'))
            ciphertext += encrypted
        else:
            ciphertext += m
    return ciphertext

def decrypt(ciphertext, keyword):
    key = generate_key(ciphertext, keyword)
    plaintext = ""
    for c, k in zip(ciphertext, key):
        if c.isalpha():
            shift = ord(k) - ord('A')
            if c.isupper():
                decrypted = chr((ord(c) - ord('A') - shift) % 26 + ord('A'))
            else:
                decrypted = chr((ord(c) - ord('a') - shift) % 26 + ord('a'))
            plaintext += decrypted
        else:
            plaintext += c

    return plaintext

from math import gcd

def mod_inverse(a, m):
    for i in range(1, m):
        if (a * i) % m == 1:
            return i
    return None

def encrypt(message, a, b):
    if gcd(a, 26) != 1:
        raise ValueError("Key 'a' must be coprime with 26.")
    ciphertext = ""
    for char in message:
        if char.isalpha():
            if char.isupper():
                x = ord(char) - ord('A')
                y = (a * x + b) % 26
                ciphertext += chr(y + ord('A'))
            else:
                x = ord(char) - ord('a')
                y = (a * x + b) % 26
                ciphertext += chr(y + ord('a'))
        else:
            ciphertext += char
    return ciphertext

def decrypt(ciphertext, a, b):
    if gcd(a, 26) != 1:
        raise ValueError("Key 'a' must be coprime with 26.")
    a_inv = mod_inverse(a, 26)
    plaintext = ""
    for char in ciphertext:
        if char.isalpha():
            if char.isupper():
                y = ord(char) - ord('A')
                x = (a_inv * (y - b)) % 26
                plaintext += chr(x + ord('A'))
            else:
                y = ord(char) - ord('a')
                x = (a_inv * (y - b)) % 26
                plaintext += chr(x + ord('a'))
        else:
            plaintext += char

    return plaintext

# Derive a 32-byte AES key from a password
def generate_key(password, salt):
    return PBKDF2(password, salt, dkLen=32)

def encrypt_file(input_file, output_file, password):
    # Generate random salt and IV
    salt = get_random_bytes(16)
    iv = get_random_bytes(16)

    # Generate encryption key
    key = generate_key(password.encode(), salt)

    # Create AES cipher
    cipher = AES.new(key, AES.MODE_CBC, iv)

    # Read file
    with open(input_file, "rb") as f:
        plaintext = f.read()

    # Encrypt
    ciphertext = cipher.encrypt(pad(plaintext, AES.block_size))

    # Save salt + IV + ciphertext
    with open(output_file, "wb") as f:
        f.write(salt)
        f.write(iv)
        f.write(ciphertext)

    print("File encrypted successfully.")

def decrypt_file(input_file, output_file, password):
    with open(input_file, "rb") as f:
        data = f.read()

    # Extract salt, IV, and ciphertext
    salt = data[:16]
    iv = data[16:32]
    ciphertext = data[32:]

    key = generate_key(password.encode(), salt)  # Generate key

    cipher = AES.new(key, AES.MODE_CBC, iv)  # Create cipher

    # Decrypt
    plaintext = unpad(cipher.decrypt(ciphertext), AES.block_size)

    # Save decrypted file
    with open(output_file, "wb") as f:
        f.write(plaintext)

    print("File decrypted successfully.")

from PIL import Image

DELIMITER = "1111111111111110"

def message_to_binary(message):
    """Convert a text message to a binary string."""
    return ''.join(format(byte, '08b') for byte in message.encode('utf-8'))

def binary_to_message(binary):
    """Convert a binary string back to text."""
    data = bytearray()

    for i in range(0, len(binary), 8):
        byte = binary[i:i + 8]
        if len(byte) == 8:
            data.append(int(byte, 2))

    return data.decode('utf-8')

def encode(input_image, output_image, message):
    """Hide a message inside an image."""
    image = Image.open(input_image).convert("RGB")

    # Convert message to binary and add an end marker
    binary_message = message_to_binary(message) + DELIMITER

    # Check whether the image has enough pixels
    capacity = image.width * image.height * 3
    if len(binary_message) > capacity:
        raise ValueError("Message is too large for this image.")

    pixels = list(image.getdata())

    data_index = 0
    new_pixels = []

    for pixel in pixels:
        new_pixel = list(pixel)

        for channel in range(3):
            if data_index < len(binary_message):
                # Clear the last bit and replace it with message bit
                new_pixel[channel] = (
                    new_pixel[channel] & 254
                ) | int(binary_message[data_index])

                data_index += 1

        new_pixels.append(tuple(new_pixel))

    image.putdata(new_pixels)
    image.save(output_image, format="PNG")

    print(f"Message successfully hidden in {output_image}")

def decode(input_image):
    """Extract a hidden message from an image."""
    image = Image.open(input_image).convert("RGB")

    binary_data = ""

    for pixel in image.getdata():
        for channel in pixel:
            binary_data += str(channel & 1)

            # Stop when the delimiter is found
            if binary_data.endswith(DELIMITER):
                binary_message = binary_data[:-len(DELIMITER)]
                return binary_to_message(binary_message)

    raise ValueError("No hidden message found.")

s.bind(('127.0.0.1', 8443))
s.listen(1)

conn, _ = s.accept()
print("[+] Connected!")

while True:
    msg = cipher.decrypt(conn.recv(4096)).decode()
    print(f"Client: {msg}")

    reply = input("Server: ")
    conn.sendall(cipher.encrypt(reply.encode()))

conn.close()

def gcd(a, b):
    return gcd(b, a % b) if b else a

def generate_keys(p, q):
    n, phi = p * q, (p - 1) * (q - 1)
    e = next(i for i in range(2, phi) if gcd(i, phi) == 1)
    d = pow(e, -1, phi)  # Built-in modular inverse (Python 3.8+)
    return (e, n), (d, n)

def add(P, Q):
    if P is None: return Q
    x, y = P; X, Y = Q
    m = ((Y-y) * pow(X-x, -1, p) if P != Q
         else (3*x*x+a) * pow(2*y, -1, p)) % p
    return ((m*m-x-X) % p, (m*(x-(m*m-x-X))-y) % p)

def mul(k, P):
    R = None
    for _ in range(k):
        R = add(R, P)
    return R
